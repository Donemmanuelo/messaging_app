# Frontend Environment Variables Example (.env)

# Backend API base URL (should point to your backend server)
NEXT_PUBLIC_API_URL=http://localhost:8000/api

# WebSocket URL (ws:// or wss:// for secure)
NEXT_PUBLIC_WS_URL=ws://localhost:8000/api/ws

# Media base URL (for accessing uploaded media)
NEXT_PUBLIC_MEDIA_URL=http://localhost:8000/media
